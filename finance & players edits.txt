// NEW — decides how much to repay when landing on the Bank (pairs with
// your existing repayLoan() in finance.c, which just executes this amount)
int decideLoanRepaymentAmount(GameState *gs, int playerIndex) {
    Player *p = &gs->players[playerIndex];
    if (!p->hasLoan) return 0;

    if (p->strategy == STRATEGY_AGGRESSIVE) {
        // "repays loans only when excess cash exceeds twice the outstanding loan amount"
        if (p->cash > 2 * p->loanAmount) {
            return p->loanAmount;
        }
        return 0;

    } else if (p->strategy == STRATEGY_CONSERVATIVE) {
        // "repays loans immediately whenever visiting the Bank if sufficient funds exist"
        if (p->cash >= p->loanAmount) {
            return p->loanAmount;
        }
        return p->cash; // partial repayment if that's all they have

    } else if (p->strategy == STRATEGY_RISK_TAKER) {
        return 0; // never voluntarily repays

    } else { // STRATEGY_OPPORTUNISTIC
        if (p->cash - p->loanAmount >= p->cash / 2) {
            return p->loanAmount;
        }
        return 0;
    }
}

// NEW — decides WHICH property (if any) to insure right now
// (pairs with buyInsurance() in finance.c, which needs a squareIndex)
int decideInsuranceTarget(GameState *gs, int playerIndex) {
    for (int i = 0; i < BOARD_SIZE; i++) {
        Square *sq = &gs->board[i];
        if (sq->ownerId == playerIndex && sq->insurancePolicy == INSURANCE_NONE) {
            if (sq->numHouses > 0 || sq->hasHotel) {
                return i; // first developed, uninsured property found
            }
        }
    }
    return -1; // nothing worth insuring right now
}

// NEW — decides WHICH policy type to buy for a given property
// (pairs with buyInsurance() in finance.c, which needs a policy type)
InsuranceType decideInsurancePolicy(GameState *gs, int playerIndex, int squareIndex) {
    Square *sq = &gs->board[squareIndex];
    Player *p = &gs->players[playerIndex];

    if (p->strategy == STRATEGY_AGGRESSIVE) {
        if (sq->hasHotel) return INSURANCE_COMPREHENSIVE;
        if (sq->numHouses > 0) return INSURANCE_BASIC;
        return INSURANCE_NONE;

    } else if (p->strategy == STRATEGY_CONSERVATIVE) {
        // "always purchases Comprehensive Insurance for every developed property"
        return INSURANCE_COMPREHENSIVE;

    } else if (p->strategy == STRATEGY_RISK_TAKER) {
        // "purchases insurance only after experiencing a financial loss" —
        // simplified stand-in: only insures once damaged already
        if (sq->isDamaged) return INSURANCE_BASIC;
        return INSURANCE_NONE;

    } else { // STRATEGY_OPPORTUNISTIC
        // "purchases Comprehensive Insurance only for high-value developments"
        if (sq->hasHotel) return INSURANCE_COMPREHENSIVE;
        return INSURANCE_NONE;
    }
}

case SQ_BANK: {
    int repayAmount = decideLoanRepaymentAmount(gs, playerIndex);   // players.c decides
    if (repayAmount > 0) {
        repayLoan(gs, playerIndex, repayAmount);                     // finance.c executes
    }
    // (your existing loan-TAKING logic stays here too, unchanged)
    break;
}

/* ============================================================
   FILE: game.c
   WHERE: inside your SQ_INSURANCE case in resolveLanding()/playTurn() —
   this REPLACES any direct call to buyInsurance() with the decide-then-execute pair
   ============================================================ */

case SQ_INSURANCE: {
    int target = decideInsuranceTarget(gs, playerIndex);                    // WHICH property
    if (target != -1) {
        InsuranceType policy = decideInsurancePolicy(gs, playerIndex, target); // WHICH policy
        if (policy != INSURANCE_NONE) {
            buyInsurance(gs, playerIndex, target, policy);                  // finance.c executes
        }
    }
    break;
}
